FILE: .github/workflows/ci.yml
========================================================================================
name: DataPulse build

on:
  push:
    branches: [main, dev]
  pull_request:
    branches: [main, dev]

jobs:
  frontend:
    name: Build Bakya frontend
    runs-on: ubuntu-latest
    defaults:
      run:
        working-directory: bakya/frontend
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with:
          node-version: 22
          cache: npm
          cache-dependency-path: bakya/frontend/package-lock.json
      - run: npm ci
      - run: npm run build

  backend:
    name: Package Udiksha backend
    runs-on: ubuntu-latest
    defaults:
      run:
        working-directory: udiksha/backend
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-java@v4
        with:
          distribution: temurin
          java-version: 21
          cache: maven
      - run: mvn -B -ntp -DskipTests package

FILE: bakya/frontend/Dockerfile
========================================================================================
FROM node:22-alpine AS build
WORKDIR /app
COPY package*.json ./
RUN npm ci
COPY . .
ARG VITE_API_URL=http://localhost:8080/api
ENV VITE_API_URL=$VITE_API_URL
RUN npm run build

FROM nginx:alpine
COPY bakya_nginx.conf /etc/nginx/conf.d/default.conf
COPY --from=build /app/dist /usr/share/nginx/html
EXPOSE 80

const API_BASE = import.meta.env.VITE_API_URL || 'http://localhost:8080/api';

async function request(path, { token, ...options } = {}) {
  const headers = new Headers(options.headers || {});
  if (token) headers.set('Authorization', `Bearer ${token}`);
  if (!(options.body instanceof FormData) && options.body !== undefined) {
    headers.set('Content-Type', 'application/json');
  }

  const response = await fetch(`${API_BASE}${path}`, { ...options, headers });
  if (response.status === 204) return null;
  const contentType = response.headers.get('content-type') || '';
  const payload = contentType.includes('application/json') ? await response.json() : await response.text();
  if (!response.ok) {
    const message = typeof payload === 'object' && payload?.message
      ? payload.message
      : `Request failed (${response.status}).`;
    const error = new Error(message);
    error.status = response.status;
    error.code = payload?.code;
    throw error;
  }
  return payload;
}

function BakyaDatasetWorkspace({ token, initialDataset, onClose, onReload, onSelectRun }) {
  const [dataset, setDataset] = useState(initialDataset);
  const [fieldName, setFieldName] = useState('');
  const [dataType, setDataType] = useState('TEXT');
  const [fieldRule, setFieldRule] = useState({});
  const [newRuleType, setNewRuleType] = useState('REQUIRED');
  const [editName, setEditName] = useState(initialDataset.datasetName);
  const [editDescription, setEditDescription] = useState(initialDataset.description || '');
  const [ruleFieldId, setRuleFieldId] = useState(null);
  const [runResult, setRunResult] = useState(null);
  const [message, setMessage] = useState('');
  const [error, setError] = useState('');
  const [saving, setSaving] = useState(false);

  useEffect(() => {
    bakyaApi.dataset(token, initialDataset.datasetId).then((fresh) => {
      setDataset(fresh); setEditName(fresh.datasetName); setEditDescription(fresh.description || '');
    }).catch((err) => setError(err.message));
  }, [token, initialDataset.datasetId]);

  async function submitField(event) {
    event.preventDefault();
    setSaving(true); setError(''); setMessage('');
    try {
      const created = await bakyaApi.addField(token, dataset.datasetId, { fieldName, dataType });
      const ruleType = fieldRule.type;
      if (ruleType) {
        const ruleConfig = ruleType === 'RANGE' ? { min: Number(fieldRule.min), max: Number(fieldRule.max) } : {};
        await bakyaApi.addRule(token, created.fieldId, { ruleType, severity: fieldRule.severity || 'MEDIUM', ruleConfig });
      }
      setFieldName(''); setFieldRule({});
      await refreshDataset();
      setMessage('Field saved.');
    } catch (err) { setError(err.message); }
    finally { setSaving(false); }
  }

  async function saveDataset(event) {
    event.preventDefault(); setSaving(true); setError(''); setMessage('');
    try {
      await bakyaApi.updateDataset(token, dataset.datasetId, { datasetName: editName, description: editDescription });
      await refreshDataset(); await onReload(); setMessage('Dataset details updated.');
    } catch (err) { setError(err.message); }
    finally { setSaving(false); }
  }

  async function renameField(field) {
    const fieldName = window.prompt('Field name', field.fieldName);
    if (!fieldName?.trim() || fieldName.trim() === field.fieldName) return;
    try {
      await bakyaApi.updateField(token, field.fieldId, { fieldName: fieldName.trim(), dataType: field.dataType });
      await refreshDataset(); setMessage('Field updated.');
    } catch (err) { setError(err.message); }
  }

  async function archiveField(field) {
    if (!window.confirm(`Archive field “${field.fieldName}”? Past validation history will be retained.`)) return;
    try { await bakyaApi.archiveField(token, field.fieldId); await refreshDataset(); setMessage('Field archived.'); }
    catch (err) { setError(err.message); }
  }

  async function addRuleToField(event) {
    event.preventDefault();
    const form = new FormData(event.currentTarget);
    const ruleType = form.get('ruleType');
    const ruleConfig = ruleType === 'RANGE' ? { min: Number(form.get('min')), max: Number(form.get('max')) } : {};
    try {
      await bakyaApi.addRule(token, ruleFieldId, { ruleType, severity: form.get('severity'), ruleConfig });
      setRuleFieldId(null); await refreshDataset(); setMessage('Rule added.');
    } catch (err) { setError(err.message); }
  }

  async function deactivateRule(rule) {
    if (!window.confirm(`Disable the ${rule.ruleType} rule? Existing run history will remain.`)) return;
    try { await bakyaApi.deactivateRule(token, rule.ruleId); await refreshDataset(); setMessage('Rule disabled.'); }
    catch (err) { setError(err.message); }
  }

  async function submitRun(event) {
    event.preventDefault();
    const file = event.currentTarget.elements.file.files[0];
    if (!file) { setError('Choose a CSV file first.'); return; }
    setSaving(true); setError(''); setMessage('');
    try {
      const result = await bakyaApi.createRun(token, dataset.datasetId, file);
      setRunResult(result);
      setMessage(`Run ${result.runId} completed with ${result.issueCount} issue(s).`);
      await onReload();
    } catch (err) { setError(err.message); }
    finally { setSaving(false); }
  }

  async function resolveIssue(issue, status) {
    const note = window.prompt(status === 'RESOLVED' ? 'Briefly describe the resolution:' : 'Why are you changing this issue?');
    if (!note?.trim()) return;
    try {
      await bakyaApi.updateIssue(token, issue.issueId, { status, note: note.trim() });
      const updated = await bakyaApi.run(token, issue.runId);
      setRunResult(updated);
      await onReload();
    } catch (err) { setError(err.message); }
  }

  async function refreshDataset() {
    const fresh = await bakyaApi.dataset(token, dataset.datasetId);
    setDataset(fresh); setEditName(fresh.datasetName); setEditDescription(fresh.description || '');
  }

  return (
    <div className="modal-backdrop" role="presentation" onMouseDown={(event) => { if (event.target === event.currentTarget) onClose(); }}>
      <section className="workspace-modal" role="dialog" aria-modal="true" aria-labelledby="dataset-title">
        <div className="panel-heading modal-title"><div><p className="eyebrow">DATASET WORKSPACE</p><h2 id="dataset-title">{dataset.datasetName}</h2><p className="muted">Configure fields, rules, and validation runs.</p></div><button className="icon-button" onClick={onClose} aria-label="Close dataset workspace">×</button></div>
        {error && <div role="alert" className="notice notice-error">{error}</div>}{message && <div className="notice notice-success">{message}</div>}
        <section className="sub-panel dataset-edit-panel"><div className="panel-heading"><div><h3>Dataset details</h3><p className="muted">Edit the name or description.</p></div></div><form className="inline-form" onSubmit={saveDataset}><label>Dataset name<input required maxLength="160" value={editName} onChange={(event) => setEditName(event.target.value)} /></label><label>Description<input maxLength="2000" value={editDescription} onChange={(event) => setEditDescription(event.target.value)} /></label><button className="button button-quiet" disabled={saving}>Save</button></form></section>
        <div className="workspace-grid">
          <section className="sub-panel"><div className="panel-heading"><div><h3>Fields & rules</h3><p className="muted">Rules run against each uploaded row.</p></div></div>
            {!dataset.fields?.length ? <p className="small-note">Add fields that match the CSV header names.</p> : <div className="field-list">{dataset.fields.map((field) => <article className="field-card" key={field.fieldId}>
              <div><strong>{field.fieldName}</strong><span className="type-label">{field.dataType}</span><span className={`status-pill ${field.active ? 'status-active' : 'status-archived'}`}>{field.active ? 'ACTIVE' : 'ARCHIVED'}</span></div>
              {field.rules?.filter((rule) => rule.active).length > 0 && <ul>{field.rules.filter((rule) => rule.active).map((rule) => <li key={rule.ruleId}>{rule.ruleType}{rule.ruleType === 'RANGE' ? ` ${rule.ruleConfig?.min}–${rule.ruleConfig?.max}` : ''}<span>{rule.severity.toLowerCase()}</span><button className="text-button danger-text" type="button" onClick={() => deactivateRule(rule)}>Disable</button></li>)}</ul>}
              {field.active && <div className="field-actions"><button className="text-button" type="button" onClick={() => renameField(field)}>Rename</button><button className="text-button" type="button" onClick={() => setRuleFieldId(ruleFieldId === field.fieldId ? null : field.fieldId)}>Add rule</button><button className="text-button danger-text" type="button" onClick={() => archiveField(field)}>Archive field</button></div>}
              {ruleFieldId === field.fieldId && <form className="form-stack compact-form" onSubmit={addRuleToField}><label>Rule<select name="ruleType" value={newRuleType} onChange={(event) => setNewRuleType(event.target.value)}><option value="REQUIRED">Required</option><option value="UNIQUE">Unique</option><option value="EMAIL">Email format</option><option value="RANGE">Number range</option><option value="TYPE">Field data type</option></select></label>{newRuleType === 'RANGE' && <div className="two-inputs"><label>Minimum<input name="min" type="number" step="any" required /></label><label>Maximum<input name="max" type="number" step="any" required /></label></div>}<label>Severity<select name="severity"><option>MEDIUM</option><option>LOW</option><option>HIGH</option></select></label><button className="button button-quiet">Save rule</button></form>}

import React, { useEffect, useState } from 'react';
import { bakyaApi } from './bakya_api.js';

const PAGE_SIZE = 10;

export default function BakyaRunDetailsPage({ token, runId, onBack }) {
  const [run, setRun] = useState(null);
  const [pageData, setPageData] = useState(null);
  const [page, setPage] = useState(0);
  const [error, setError] = useState('');

  async function load() {
    try {
      const [runResult, issuesResult] = await Promise.all([
        bakyaApi.run(token, runId),
        bakyaApi.issues(token, runId, { page, size: PAGE_SIZE }),
      ]);
      setRun(runResult);
      setPageData(issuesResult);
      setError('');
    } catch (err) { setError(err.message); }
  }

  useEffect(() => { load(); }, [token, runId, page]);

  async function changeStatus(issue, status) {
    const note = window.prompt(status === 'RESOLVED' ? 'Briefly describe the resolution:' : 'Why are you changing this issue?');
    if (!note?.trim()) return;
    try {
      await bakyaApi.updateIssue(token, issue.issueId, { status, note: note.trim() });
      await load();
    } catch (err) { setError(err.message); }
  }

  if (error && !run) return <div className="page-stack"><button className="button button-quiet" onClick={onBack}>← Back</button><div role="alert" className="notice notice-error">{error}</div></div>;
  if (!run || !pageData) return <div className="loading-state">Loading validation run…</div>;

  return <div className="page-stack">
    <div className="page-heading"><div><p className="eyebrow">VALIDATION RUN / {run.runId}</p><h1>{run.datasetName}</h1><p className="muted">Started {new Date(run.startedAt).toLocaleString()} · {run.totalRows} rows · {run.issueCount} issues</p></div><button className="button button-quiet" onClick={onBack}>← Back to datasets</button></div>
    {error && <div role="alert" className="notice notice-error">{error}</div>}
    <section className="metric-grid"><article className="metric-card"><span className="metric-label">Overall quality</span><strong>{run.score ? <>{Number(run.score.overallScore).toFixed(1)}<em>/100</em></> : '—'}</strong><small>Completed {run.completedAt ? new Date(run.completedAt).toLocaleString() : 'in progress'}</small></article><article className="metric-card"><span className="metric-label">Rows checked</span><strong>{run.totalRows}</strong><small>CSV data records</small></article><article className="metric-card"><span className="metric-label">Open issues</span><strong>{pageData.content.filter((issue) => issue.status === 'OPEN').length}</strong><small>{run.issueCount} total issues across all pages</small></article></section>
    <section className="panel"><div className="panel-heading"><div><h2>Validation issues</h2><p className="muted">Resolution actions are recorded in the audit history.</p></div><span className="count-tag">{pageData.totalElements} issues</span></div>
      {!pageData.content.length ? <div className="empty-state"><h3>No issues found</h3><p>This run passed every configured check.</p></div> : <div className="table-scroll"><table><thead><tr><th>Row</th><th>Field</th><th>Severity</th><th>Issue</th><th>Observed value</th><th>Status</th><th>Action</th></tr></thead><tbody>{pageData.content.map((issue) => <tr key={issue.issueId}>
        <td>{issue.rowNumber}</td><td>{issue.fieldName}</td><td><span className={`severity severity-${issue.severity.toLowerCase()}`}>{issue.severity}</span></td><td>{issue.message}</td><td>{issue.observedValue || '—'}</td><td><span className={`status-pill ${issue.status === 'OPEN' ? 'status-open' : 'status-done'}`}>{issue.status}</span></td><td>{issue.status === 'OPEN' ? <button className="text-button" onClick={() => changeStatus(issue, 'RESOLVED')}>Resolve</button> : <button className="text-button" onClick={() => changeStatus(issue, 'OPEN')}>Reopen</button>}</td>
      </tr>)}</tbody></table></div>}
      <div className="pagination"><span>Page {pageData.page + 1} of {Math.max(pageData.totalPages, 1)}</span><div><button className="button button-quiet" disabled={page <= 0} onClick={() => setPage((value) => Math.max(0, value - 1))}>Previous</button><button className="button button-quiet" disabled={page + 1 >= pageData.totalPages} onClick={() => setPage((value) => value + 1)}>Next</button></div></div>
    </section>
  </div>;
}

# DataPulse

**DataPulse** is a small data-quality management platform for institutional datasets. An Analyst defines dataset fields and validation rules, uploads a CSV to run those rules, reviews quality issues and scores, and records issue resolutions. Admins can manage all datasets; Analysts manage their own.

## Team ownership

- `bakya/frontend/` — React UI, owned by Bakya (24BAI0071). Authored frontend source and assets use the `bakya_` prefix where practical; build-required filenames such as `package.json` stay standard.
- `udiksha/backend/` — Spring Boot REST API, JPA/Hibernate, JWT/BCrypt and validation workflow, owned by Udiksha (24BAI0086).
- `rudra/db/` and `rudra/docs/` — PostgreSQL schema, ERD, data dictionary, API contract, assessment checklist and SDG text, owned by Rudra (24BKT0125).

## Requirements

- Java 21 and Maven 3.6.3 or later
- Node.js 22.12+ and npm
- Docker Desktop with Compose, or PostgreSQL 18 and a local database

The project uses Spring Boot 3.5.16, React 19.3, and Vite 7. React/Vite version requirements are documented in the official [React release notes](https://react.dev/blog/2026/09/09/react-19-3) and [Vite guide](https://vite.dev/guide/); Spring Boot 3.5 requires Java 17 or later according to its [system requirements](https://docs.spring.io/spring-boot/3.5/system-requirements.html). The project targets Java 21.

## Quick start with Docker

1. Copy `.env.example` to `.env` and replace the database password, JWT secret, and local demo passwords. Generate a JWT secret with `openssl rand -base64 32`.
2. Run `docker compose up --build` from this folder.
3. Open the frontend at `http://localhost:5173` and the API health endpoint at `http://localhost:8080/api/health`.
4. Use the Admin and Analyst credentials configured in `.env`.

The PostgreSQL initialization script creates the schema and the ten related tables on the first database start. The Postgres 18 image stores its data under `/var/lib/postgresql`, so Compose mounts the named volume at that image path, as described in the [official Postgres image notes](https://github.com/docker-library/docs/blob/master/postgres/README.md#pgdata). If you change the schema and need a clean local database, stop the stack and remove its named volume with `docker compose down -v`, then start it again. This deletes local database contents.

## Run services separately

1. Start PostgreSQL and apply `rudra/db/schema.sql` to a new `datapulse` database.
2. Copy `.env.example` to `.env` and replace every `REPLACE_...` value. Spring Boot does not read `.env` automatically; export those values in the shell before starting Maven, or use Docker Compose.
3. From `udiksha/backend`, run `mvn spring-boot:run`.
4. From `bakya/frontend`, run `npm install` once and then `npm run dev`.
5. Set `VITE_API_URL=http://localhost:8080/api` for the frontend and configure the backend variables shown in `.env.example`.

## Demo workflow

1. Log in as an Analyst and create a dataset with `record_id`, `email`, `attendance_pct`, and `department` fields.
2. Add a required and unique rule for `record_id`, required and email rules for `email`, and a range rule from 0 to 100 for `attendance_pct`.
3. Upload `rudra/db/sample_validation.csv` from the dataset's run screen.
4. Review invalid email, duplicate ID, out-of-range attendance and missing email issues; resolve one issue.
5. Log in as Admin and demonstrate role-gated access.

See `rudra/docs/PROJECT_PROPOSAL.md` for the proposal and SDG alignment, `rudra/docs/API_CONTRACT.md` for the frontend/backend contract, and `rudra/docs/FINAL_SUBMISSION_CHECKLIST.md` for the assessment evidence still to capture.
Use `rudra/docs/GITHUB_HANDOFF.md` for repo setup and `rudra/docs/FINAL_SUBMISSION_GUIDE.md` for member-specific PDF and upload requirements.

## Database rubric map

`rudra/db/schema.sql` defines primary and foreign keys, NOT NULL/UNIQUE/CHECK/DEFAULT constraints, ten normalized tables, indexes, `dataset_quality_summary` view, `recalculate_quality_score` stored procedure, and an audit trigger. The backend uses JPA/Hibernate for routine CRUD. The run service must call the procedure in the transaction that persists the run and its issues.

## Backups

With the Compose stack running, create a native PostgreSQL backup with `docker compose exec -T db sh -c 'pg_dump -U "$POSTGRES_USER" -d "$POSTGRES_DB" -Fc' > datapulse.dump`. Create an empty restore database with `docker compose exec -T db sh -c 'createdb -U "$POSTGRES_USER" datapulse_restore'`, then restore it with `docker compose exec -T db sh -c 'pg_restore -U "$POSTGRES_USER" -d datapulse_restore --clean --if-exists' < datapulse.dump`. Check the restored schema and data before recording backup/restore evidence. Do not commit database dumps containing real data.

Add the final live URL, actual registration numbers, ERD image, complete source code text, and genuine timestamped screenshots to the assessment package before upload.

-- DataPulse PostgreSQL schema. Run in a new project database.
-- The backend should use JPA/Hibernate for ordinary data access. This SQL file
-- defines the advanced database objects required by the assessment.

BEGIN;

CREATE TABLE roles (
    role_id       bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    role_name     varchar(30) NOT NULL UNIQUE,
    description   text,
    CONSTRAINT ck_roles_role_name CHECK (role_name IN ('ADMIN', 'ANALYST'))
);

CREATE TABLE users (
    user_id       bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    role_id       bigint NOT NULL REFERENCES roles(role_id) ON DELETE RESTRICT,
    full_name     varchar(120) NOT NULL,
    email         varchar(254) NOT NULL,
    password_hash varchar(255) NOT NULL,
    is_active     boolean NOT NULL DEFAULT true,
    created_at    timestamptz NOT NULL DEFAULT now(),
    updated_at    timestamptz NOT NULL DEFAULT now(),
    CONSTRAINT ck_users_email_nonempty CHECK (length(trim(email)) > 0),
    CONSTRAINT ck_users_full_name_nonempty CHECK (length(trim(full_name)) > 0)
);
CREATE UNIQUE INDEX ux_users_email_lower ON users (lower(email));

CREATE TABLE datasets (
    dataset_id   bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    owner_id     bigint NOT NULL REFERENCES users(user_id) ON DELETE RESTRICT,
    dataset_name varchar(160) NOT NULL,
    description  text,
    status       varchar(20) NOT NULL DEFAULT 'ACTIVE',
    created_at   timestamptz NOT NULL DEFAULT now(),
    updated_at   timestamptz NOT NULL DEFAULT now(),
    CONSTRAINT uq_datasets_owner_name UNIQUE (owner_id, dataset_name),
    CONSTRAINT ck_datasets_status CHECK (status IN ('ACTIVE', 'ARCHIVED')),
    CONSTRAINT ck_datasets_name_nonempty CHECK (length(trim(dataset_name)) > 0)
);

CREATE TABLE dataset_fields (
    field_id    bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    dataset_id  bigint NOT NULL REFERENCES datasets(dataset_id) ON DELETE CASCADE,
    field_name  varchar(120) NOT NULL,
    data_type   varchar(20) NOT NULL DEFAULT 'TEXT',
    is_active   boolean NOT NULL DEFAULT true,
    created_at  timestamptz NOT NULL DEFAULT now(),
    CONSTRAINT uq_dataset_fields_dataset_name UNIQUE (dataset_id, field_name),
    CONSTRAINT ck_dataset_fields_data_type
        CHECK (data_type IN ('TEXT', 'INTEGER', 'DECIMAL', 'DATE', 'BOOLEAN')),
    CONSTRAINT ck_dataset_fields_name_nonempty CHECK (length(trim(field_name)) > 0)
);

CREATE TABLE validation_rules (
    rule_id     bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    field_id    bigint NOT NULL REFERENCES dataset_fields(field_id) ON DELETE RESTRICT,
    rule_type   varchar(20) NOT NULL,
    severity    varchar(10) NOT NULL DEFAULT 'MEDIUM',
    rule_config jsonb NOT NULL DEFAULT '{}'::jsonb,
    is_active   boolean NOT NULL DEFAULT true,
    created_at  timestamptz NOT NULL DEFAULT now(),
    CONSTRAINT ck_validation_rules_type
        CHECK (rule_type IN ('REQUIRED', 'UNIQUE', 'EMAIL', 'RANGE', 'TYPE')),
    CONSTRAINT ck_validation_rules_severity
        CHECK (severity IN ('LOW', 'MEDIUM', 'HIGH')),
    CONSTRAINT ck_validation_rules_config_object
        CHECK (jsonb_typeof(rule_config) = 'object')
);

CREATE TABLE validation_runs (
    run_id          bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    dataset_id      bigint NOT NULL REFERENCES datasets(dataset_id) ON DELETE RESTRICT,
    started_by      bigint NOT NULL REFERENCES users(user_id) ON DELETE RESTRICT,
    started_at      timestamptz NOT NULL DEFAULT now(),
    completed_at    timestamptz,
    status          varchar(20) NOT NULL DEFAULT 'QUEUED',
    total_rows      integer NOT NULL DEFAULT 0,
    issue_count     integer NOT NULL DEFAULT 0,
    failure_message text,
    CONSTRAINT ck_validation_runs_status
        CHECK (status IN ('QUEUED', 'RUNNING', 'COMPLETED', 'FAILED')),
    CONSTRAINT ck_validation_runs_counts CHECK (total_rows >= 0 AND issue_count >= 0),
    CONSTRAINT ck_validation_runs_completed_time
        CHECK (status <> 'COMPLETED' OR completed_at IS NOT NULL)
);

CREATE TABLE quality_issues (
    issue_id       bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    run_id         bigint NOT NULL REFERENCES validation_runs(run_id) ON DELETE RESTRICT,
    rule_id        bigint NOT NULL REFERENCES validation_rules(rule_id) ON DELETE RESTRICT,
    row_number     integer NOT NULL,
    severity       varchar(10) NOT NULL,
    message        text NOT NULL,
    observed_value text,
    status         varchar(20) NOT NULL DEFAULT 'OPEN',
    created_at     timestamptz NOT NULL DEFAULT now(),
    resolved_at    timestamptz,
    CONSTRAINT uq_quality_issues_run_rule_row UNIQUE (run_id, rule_id, row_number),
    CONSTRAINT ck_quality_issues_row_number CHECK (row_number > 0),
    CONSTRAINT ck_quality_issues_severity CHECK (severity IN ('LOW', 'MEDIUM', 'HIGH')),
    CONSTRAINT ck_quality_issues_status CHECK (status IN ('OPEN', 'RESOLVED', 'WAIVED')),
    CONSTRAINT ck_quality_issues_resolution_time
        CHECK ((status = 'OPEN' AND resolved_at IS NULL) OR
               (status <> 'OPEN' AND resolved_at IS NOT NULL))
);

CREATE TABLE issue_resolutions (
    resolution_id      bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    issue_id           bigint NOT NULL REFERENCES quality_issues(issue_id) ON DELETE RESTRICT,
    resolved_by_user_id bigint NOT NULL REFERENCES users(user_id) ON DELETE RESTRICT,
    action             varchar(20) NOT NULL,
    note               text NOT NULL,
    resolved_at        timestamptz NOT NULL DEFAULT now(),
    CONSTRAINT ck_issue_resolutions_action
        CHECK (action IN ('FIXED', 'WAIVED', 'REOPENED')),
    CONSTRAINT ck_issue_resolutions_note_nonempty CHECK (length(trim(note)) > 0)
);

CREATE TABLE quality_scores (
    score_id            bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    run_id              bigint NOT NULL UNIQUE REFERENCES validation_runs(run_id) ON DELETE RESTRICT,
    completeness_score  numeric(5,2) NOT NULL,
    validity_score      numeric(5,2) NOT NULL,
    uniqueness_score    numeric(5,2) NOT NULL,
    overall_score       numeric(5,2) NOT NULL,
    calculated_at       timestamptz NOT NULL DEFAULT now(),
    CONSTRAINT ck_quality_scores_range CHECK (
        completeness_score BETWEEN 0 AND 100 AND
        validity_score BETWEEN 0 AND 100 AND
        uniqueness_score BETWEEN 0 AND 100 AND
        overall_score BETWEEN 0 AND 100
    )
);

CREATE TABLE audit_logs (
    log_id        bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    actor_user_id bigint REFERENCES users(user_id) ON DELETE SET NULL,
    entity_type   varchar(40) NOT NULL,
    entity_id     bigint NOT NULL,
    action        varchar(40) NOT NULL,
    old_values    jsonb,
    new_values    jsonb,
    created_at    timestamptz NOT NULL DEFAULT now()
);

-- Indexes support the app's common filters, joins, and history pages.
CREATE INDEX ix_datasets_owner_status ON datasets(owner_id, status);
CREATE INDEX ix_dataset_fields_dataset ON dataset_fields(dataset_id);
CREATE INDEX ix_validation_rules_field_active ON validation_rules(field_id, is_active);
CREATE INDEX ix_validation_runs_dataset_started ON validation_runs(dataset_id, started_at DESC);
CREATE INDEX ix_quality_issues_run_status_severity ON quality_issues(run_id, status, severity);
CREATE INDEX ix_issue_resolutions_issue_time ON issue_resolutions(issue_id, resolved_at DESC);
CREATE INDEX ix_audit_logs_entity_time ON audit_logs(entity_type, entity_id, created_at DESC);

-- Summary view used by the dashboard: one latest completed score and current
-- open-issue count per dataset.
CREATE VIEW dataset_quality_summary AS
SELECT
    d.dataset_id,
    d.dataset_name,
    d.status AS dataset_status,
    latest.run_id AS latest_run_id,
    latest.overall_score AS latest_overall_score,
    COALESCE(open_issues.open_issue_count, 0)::bigint AS open_issue_count
FROM datasets AS d
LEFT JOIN LATERAL (
    SELECT vr.run_id, qs.overall_score
    FROM validation_runs AS vr
    JOIN quality_scores AS qs ON qs.run_id = vr.run_id
    WHERE vr.dataset_id = d.dataset_id AND vr.status = 'COMPLETED'
    ORDER BY vr.completed_at DESC, vr.run_id DESC
    LIMIT 1
) AS latest ON true
LEFT JOIN LATERAL (
    SELECT count(*) AS open_issue_count
    FROM quality_issues AS qi
    JOIN validation_runs AS vr ON vr.run_id = qi.run_id
    WHERE vr.dataset_id = d.dataset_id AND qi.status = 'OPEN'
) AS open_issues ON true;

-- Score procedure. The application must call this inside the same transaction
-- that finalizes the run and writes its issues.
CREATE PROCEDURE recalculate_quality_score(IN p_run_id bigint)
LANGUAGE plpgsql
AS $$
DECLARE
    v_dataset_id       bigint;
    v_total_rows       bigint;
    v_field_count      bigint;
    v_total_cells      bigint;
    v_missing_cells    bigint;
    v_duplicate_cells  bigint;
    v_invalid_cells    bigint;
    v_completeness     numeric(5,2);
    v_validity         numeric(5,2);
    v_uniqueness       numeric(5,2);
    v_overall          numeric(5,2);
BEGIN
    SELECT dataset_id, total_rows
      INTO v_dataset_id, v_total_rows
      FROM validation_runs
     WHERE run_id = p_run_id;

    IF NOT FOUND THEN
        RAISE EXCEPTION 'Validation run % does not exist', p_run_id;
    END IF;

    SELECT count(*) INTO v_field_count
      FROM dataset_fields
     WHERE dataset_id = v_dataset_id AND is_active = true;

    v_total_cells := v_total_rows * v_field_count;

    SELECT count(DISTINCT (qi.row_number, vr.field_id))
      INTO v_missing_cells
      FROM quality_issues AS qi
      JOIN validation_rules AS vr ON vr.rule_id = qi.rule_id
     WHERE qi.run_id = p_run_id AND vr.rule_type = 'REQUIRED';

    SELECT count(DISTINCT (qi.row_number, vr.field_id))
      INTO v_duplicate_cells
      FROM quality_issues AS qi
      JOIN validation_rules AS vr ON vr.rule_id = qi.rule_id
     WHERE qi.run_id = p_run_id AND vr.rule_type = 'UNIQUE';

    SELECT count(DISTINCT (qi.row_number, vr.field_id))
      INTO v_invalid_cells
      FROM quality_issues AS qi
      JOIN validation_rules AS vr ON vr.rule_id = qi.rule_id
     WHERE qi.run_id = p_run_id;

    IF v_total_cells = 0 THEN
        v_completeness := 0;
        v_validity := 0;
        v_uniqueness := 0;
    ELSE
        v_completeness := round(100.0 * (v_total_cells - LEAST(v_missing_cells, v_total_cells)) / v_total_cells, 2);
        v_validity := round(100.0 * (v_total_cells - LEAST(v_invalid_cells, v_total_cells)) / v_total_cells, 2);
        v_uniqueness := round(100.0 * (v_total_cells - LEAST(v_duplicate_cells, v_total_cells)) / v_total_cells, 2);
    END IF;

    v_overall := round((v_completeness + v_validity + v_uniqueness) / 3.0, 2);

    INSERT INTO quality_scores (
        run_id, completeness_score, validity_score, uniqueness_score, overall_score, calculated_at
    ) VALUES (
        p_run_id, v_completeness, v_validity, v_uniqueness, v_overall, now()
    )
    ON CONFLICT (run_id) DO UPDATE SET
        completeness_score = EXCLUDED.completeness_score,
        validity_score = EXCLUDED.validity_score,
        uniqueness_score = EXCLUDED.uniqueness_score,
        overall_score = EXCLUDED.overall_score,
        calculated_at = EXCLUDED.calculated_at;
END;
$$;

-- Audit issue creation and status transitions. The backend may set a transaction
-- local value with: SET LOCAL app.user_id = '<authenticated user ID>'.
CREATE FUNCTION audit_quality_issue_change()
RETURNS trigger
LANGUAGE plpgsql
AS $$
DECLARE
    v_actor_user_id bigint;
BEGIN
    v_actor_user_id := NULLIF(current_setting('app.user_id', true), '')::bigint;

    IF TG_OP = 'INSERT' THEN
        INSERT INTO audit_logs (
            actor_user_id, entity_type, entity_id, action, old_values, new_values
        ) VALUES (
            v_actor_user_id, 'QUALITY_ISSUE', NEW.issue_id, 'ISSUE_CREATED', NULL,
            jsonb_build_object('status', NEW.status, 'run_id', NEW.run_id, 'rule_id', NEW.rule_id)
        );
    ELSIF OLD.status IS DISTINCT FROM NEW.status THEN
        INSERT INTO audit_logs (
            actor_user_id, entity_type, entity_id, action, old_values, new_values
        ) VALUES (
            v_actor_user_id, 'QUALITY_ISSUE', NEW.issue_id, 'ISSUE_STATUS_CHANGED',
            jsonb_build_object('status', OLD.status), jsonb_build_object('status', NEW.status)
        );
    END IF;

    RETURN NEW;
END;
$$;

CREATE TRIGGER trg_audit_quality_issue_change
AFTER INSERT OR UPDATE ON quality_issues
FOR EACH ROW
EXECUTE FUNCTION audit_quality_issue_change();

INSERT INTO roles (role_name, description) VALUES
    ('ADMIN', 'Manages users and all datasets.'),
    ('ANALYST', 'Manages owned datasets and validation runs.')
ON CONFLICT (role_name) DO NOTHING;

COMMIT;

# DataPulse API Contract

Backend: Spring Boot REST API. Frontend: React in `bakya/frontend/`. Use JSON for API bodies except the CSV run upload. Unless stated otherwise, all endpoints are under `/api` and require a bearer JWT after login.

## Authentication

| Method and path | Request | Response / behavior |
|---|---|---|
| `POST /auth/login` | `{ "email": "...", "password": "..." }` | `{ "token": "...", "user": { "userId": 1, "fullName": "...", "email": "...", "role": "ANALYST" } }` |
| `GET /auth/me` | Bearer token | Returns the authenticated user's safe profile and role. |
| `POST /auth/logout` | Bearer token | Stateless JWT clients clear the token; backend may revoke it if a denylist is implemented. |

Never return `password_hash` from an endpoint. Seed development users with BCrypt hashes and document their local-only setup without committing secrets.

## Main resources

| Method and path | Purpose |
|---|---|
| `GET /datasets?page=&size=&q=&status=` | List the user's permitted datasets with pagination and filters. Admin may view all. |
| `POST /datasets` | Create a dataset. Owner is taken from the authenticated user, not trusted from the request body. |
| `GET /datasets/{datasetId}` | Dataset details including fields and rules. |
| `PUT /datasets/{datasetId}` | Update dataset metadata. |
| `DELETE /datasets/{datasetId}` | Archive the dataset; preserve historical runs. |
| `POST /datasets/{datasetId}/fields` | Add a field with `fieldName` and `dataType`; requiredness is a `REQUIRED` validation rule. |
| `PUT /fields/{fieldId}` / `DELETE /fields/{fieldId}` | Update a field or disable it if it has history. |
| `POST /fields/{fieldId}/rules` | Add a rule with `ruleType`, `severity`, and `ruleConfig`. Types: `REQUIRED`, `UNIQUE`, `EMAIL`, `RANGE`, `TYPE`; `TYPE` checks the field's configured data type. |
| `PUT /rules/{ruleId}` / `DELETE /rules/{ruleId}` | Update or deactivate a rule. |
| `POST /datasets/{datasetId}/runs` | Multipart upload with part name `file`; validate CSV columns against dataset fields and persist run/issues/score in one transaction. |
| `GET /runs/{runId}` | Run status, counts, score, and paginated issues. |
| `PATCH /issues/{issueId}/status` | `{ "status": "RESOLVED", "note": "..." }`; append an `Issue_Resolutions` history row. |
| `GET /dashboard/summary` | Summary counts and latest quality scores using the database view/query. |
| `GET /admin/overview` | Admin-only workspace overview across every dataset; an Analyst receives HTTP 403. |

## Roles and error shape

- `ADMIN`: view and manage all datasets; access the admin overview.
- `ANALYST`: manage owned datasets, fields, rules, runs, and issue resolutions.
- Enforce ownership and roles on the API, not only by hiding frontend buttons.
- Return standard HTTP statuses and a consistent error body, for example `{ "code": "VALIDATION_ERROR", "message": "...", "details": [] }`.
- Validate page sizes, CSV headers, rule configuration, and all user-provided IDs. Never build SQL by concatenating user input.

## End-to-end acceptance workflow

1. Log in as an Analyst and create a dataset with fields `record_id`, `email`, `attendance_pct`, and `department`.
2. Add rules: `record_id` unique and required, `email` required and email-formatted, `attendance_pct` in the range 0 to 100.
3. Upload `rudra/db/sample_validation.csv` to start a run.
4. See the run summary, issues, and score; resolve at least one issue and see the updated status/history.
5. Log in as Admin and verify the admin-only page/API behavior.

# DataPulse Data Dictionary

All timestamps use PostgreSQL `timestamptz` in UTC. IDs are generated identity primary keys. The schema is implemented in `db/schema.sql`.

| Table | Column | Type | Meaning / rule |
|---|---|---|---|
| `roles` | `role_id` | `bigint` | Primary key. |
|  | `role_name` | `varchar(30)` | Unique; `ADMIN` or `ANALYST`. |
|  | `description` | `text` | Optional role description. |
| `users` | `user_id` | `bigint` | Primary key. |
|  | `role_id` | `bigint` | Required FK to `roles`. |
|  | `full_name` | `varchar(120)` | Required display name. |
|  | `email` | `varchar(254)` | Required; unique case-insensitively. |
|  | `password_hash` | `varchar(255)` | Required BCrypt hash; never plaintext. |
|  | `is_active` | `boolean` | Required; defaults true. |
|  | `created_at`, `updated_at` | `timestamptz` | Account timestamps. |
| `datasets` | `dataset_id` | `bigint` | Primary key. |
|  | `owner_id` | `bigint` | Required FK to `users`. |
|  | `dataset_name` | `varchar(160)` | Required; unique per owner. |
|  | `description` | `text` | Optional description. |
|  | `status` | `varchar(20)` | `ACTIVE` or `ARCHIVED`; defaults `ACTIVE`. |
|  | `created_at`, `updated_at` | `timestamptz` | Dataset timestamps. |
| `dataset_fields` | `field_id` | `bigint` | Primary key. |
|  | `dataset_id` | `bigint` | Required FK to `datasets`, cascades only before history exists. |
|  | `field_name` | `varchar(120)` | Required; unique within a dataset. |
|  | `data_type` | `varchar(20)` | `TEXT`, `INTEGER`, `DECIMAL`, `DATE`, or `BOOLEAN`. |
|  | `is_active` | `boolean` | Required; defaults true. Archived fields are excluded from new runs but retained for historical issues. |
|  | `created_at` | `timestamptz` | Field creation time. |
| `validation_rules` | `rule_id` | `bigint` | Primary key. |
|  | `field_id` | `bigint` | Required FK to `dataset_fields`. |
|  | `rule_type` | `varchar(20)` | `REQUIRED`, `UNIQUE`, `EMAIL`, `RANGE`, or `TYPE` (checks the field's configured type). |
|  | `severity` | `varchar(10)` | `LOW`, `MEDIUM`, or `HIGH`. |
|  | `rule_config` | `jsonb` | Type-specific settings; `RANGE` uses `min`/`max`. |
|  | `is_active` | `boolean` | Defaults true; deactivate historical rules rather than delete. |
|  | `created_at` | `timestamptz` | Rule creation time. |
| `validation_runs` | `run_id` | `bigint` | Primary key. |
|  | `dataset_id` | `bigint` | Required FK to `datasets`. |
|  | `started_by` | `bigint` | Required FK to `users`. |

# DataPulse Entity Relationship Design

DataPulse is scoped for a small, demonstrable institutional dataset workflow. An analyst registers a dataset and its fields, attaches validation rules, uploads a CSV for a run, reviews the resulting issues, and records resolution history. The CSV is processed by the backend for the run; the first version stores dataset metadata and results rather than arbitrary source rows.

## Design decisions

- There are ten tables. `Users` is the only account table; `Roles` stores role names and `Users.role_id` assigns a role. There are no separate Admin or Analyst account tables.
- Each table has a primary key. Foreign keys express ownership and history. Dataset/field/rule/run relationships are represented once; `Quality_Scores` references a run and does not duplicate its dataset ID.
- A quality issue references the validation rule that failed. Its field is obtained through `Validation_Rules.field_id`, avoiding another copy of the same relationship.
- Requiredness is represented by an active `REQUIRED` validation rule rather than duplicated as a second flag on `Dataset_Fields`.
- `Quality_Issues` also stores the CSV data-row number (header excluded), observed value, severity snapshot, message, and current status.
- The backend must verify that each run and its rules belong to the same dataset before writing results. This cross-table rule is enforced in the transactional service.
- Dataset deletion is restricted once history exists. Fields and rules are soft-archived with `is_active = false`, preserving run history and issue foreign keys.

## Cardinalities

- One role can be assigned to many users; each user has one role.
- One dataset has many fields and validation runs.
- One field can have many validation rules.
- One validation run can produce many issues; each issue is tied to one run and one rule.
- One issue can have multiple resolution-history entries.
- A completed run has at most one score record.

## Relational schema and 3NF notes

The complete relational schema is in `rudra/db/schema.sql`; the column-by-column definitions are in `rudra/docs/DATA_DICTIONARY.md`. The ten base tables are ROLES, USERS, DATASETS, DATASET_FIELDS, VALIDATION_RULES, VALIDATION_RUNS, QUALITY_ISSUES, ISSUE_RESOLUTIONS, QUALITY_SCORES, and AUDIT_LOGS. Primary and foreign keys in the ER diagram match the DDL.

The design targets Third Normal Form. Each table stores one kind of entity or event, each non-key column describes that table's key, and no non-key column is included to carry a transitive dependency from another table. Role names are stored once and referenced by USERS; datasets are referenced by fields and runs; issues reference a validation rule rather than duplicating its field; scores reference a run rather than duplicating its dataset; resolution history is kept in its own table. Severity and message on an issue are intentional run-time snapshots so later rule edits do not rewrite historical outcomes.

Integrity rules include primary/foreign keys, non-null columns, unique lowercased user email, per-owner dataset-name uniqueness, per-dataset field-name uniqueness, check constraints for statuses/rule types/scores, defaults for lifecycle columns, and indexes for common joins and filters. The backend uses ORM-based parameter binding for routine operations; raw SQL is limited to the dashboard's view/aggregate and invocation of the stored procedure.

# DataPulse — Faculty Project Proposal

## Project title

**DataPulse: Institutional Dataset Quality and Validation Platform**

## Problem statement

University departments exchange spreadsheets for routine work, but missing values, malformed entries, duplicate identifiers, and out-of-range measurements are often discovered late. Manual review is slow and leaves no consistent history of which checks ran or how issues were handled. DataPulse provides a small, auditable workflow for defining dataset fields and validation rules, checking CSV files, reviewing data-quality results, and recording issue resolutions.

## Objectives

- Store users, roles, datasets, fields, validation rules, validation runs, quality issues, resolution history, quality scores, and audit events in a normalized relational database.
- Provide a responsive web application for managing datasets and reviewing validation results.
- Apply secure authentication, role-based access, request validation, and protected database credentials.
- Demonstrate relational database features including CRUD, joins, aggregates, indexes, a view, a stored procedure, a trigger, and a transaction.
- Preserve validation history without storing uploaded CSV rows in the first project version.

## Scope

An Analyst creates a dataset, defines its columns, attaches REQUIRED, UNIQUE, EMAIL, RANGE, or configured TYPE rules, and uploads a CSV file. DataPulse checks the configured columns, saves a run and any issues, calculates a score, and allows authorized users to resolve or reopen issues with a note. Admins can see all datasets; Analysts can manage their own. The initial version processes UTF-8 CSV files up to 5 MB and 5,000 rows. It does not ingest spreadsheets or store original records.

import org.springframework.http.HttpStatus;

public class ApiException extends RuntimeException {
    private final HttpStatus status;
    private final String code;

    public ApiException(HttpStatus status, String code, String message) {
        super(message);
        this.status = status;
        this.code = code;
    }

    public HttpStatus getStatus() { return status; }
    public String getCode() { return code; }

    public static ApiException notFound(String message) {
        return new ApiException(HttpStatus.NOT_FOUND, "NOT_FOUND", message);
    }

    public static ApiException badRequest(String message) {
        return new ApiException(HttpStatus.BAD_REQUEST, "BAD_REQUEST", message);
    }

    public static ApiException forbidden(String message) {
        return new ApiException(HttpStatus.FORBIDDEN, "FORBIDDEN", message);
    }

    public static ApiException unauthorized(String message) {
        return new ApiException(HttpStatus.UNAUTHORIZED, "UNAUTHORIZED", message);
    }
}

package com.datapulse.api;

import jakarta.validation.ConstraintViolationException;
import org.springframework.dao.DataIntegrityViolationException;
import org.springframework.http.HttpStatus;
import org.springframework.http.ResponseEntity;
import org.springframework.security.access.AccessDeniedException;
import org.springframework.web.bind.MethodArgumentNotValidException;
import org.springframework.web.bind.annotation.ExceptionHandler;
import org.springframework.web.bind.annotation.RestControllerAdvice;
import org.springframework.web.multipart.MaxUploadSizeExceededException;

import java.util.LinkedHashMap;
import java.util.Map;

@RestControllerAdvice
public class GlobalExceptionHandler {
    @ExceptionHandler(ApiException.class)
    ResponseEntity<ApiDtos.ApiError> handleApi(ApiException ex) {
        return ResponseEntity.status(ex.getStatus())
                .body(new ApiDtos.ApiError(ex.getCode(), ex.getMessage(), Map.of()));
    }

    @ExceptionHandler(MethodArgumentNotValidException.class)
    ResponseEntity<ApiDtos.ApiError> handleValidation(MethodArgumentNotValidException ex) {
        Map<String, String> details = new LinkedHashMap<>();
        ex.getBindingResult().getFieldErrors().forEach(error ->
                details.putIfAbsent(error.getField(), error.getDefaultMessage()));
        return ResponseEntity.badRequest().body(
                new ApiDtos.ApiError("VALIDATION_ERROR", "Please correct the highlighted fields.", details));
    }

    @ExceptionHandler(ConstraintViolationException.class)
    ResponseEntity<ApiDtos.ApiError> handleConstraint(ConstraintViolationException ex) {
        return ResponseEntity.badRequest().body(
                new ApiDtos.ApiError("VALIDATION_ERROR", "A request constraint was not met.", Map.of()));
    }

    @ExceptionHandler(AccessDeniedException.class)
    ResponseEntity<ApiDtos.ApiError> handleAccessDenied() {
        return ResponseEntity.status(HttpStatus.FORBIDDEN).body(
                new ApiDtos.ApiError("FORBIDDEN", "You do not have access to this resource.", Map.of()));
    }

    @ExceptionHandler(DataIntegrityViolationException.class)
    ResponseEntity<ApiDtos.ApiError> handleIntegrity() {
        return ResponseEntity.status(HttpStatus.CONFLICT).body(
                new ApiDtos.ApiError("DATA_CONFLICT", "This change conflicts with existing records or a database rule.", Map.of()));
    }

    @ExceptionHandler(MaxUploadSizeExceededException.class)
    ResponseEntity<ApiDtos.ApiError> handleUploadTooLarge() {
        return ResponseEntity.badRequest().body(
                new ApiDtos.ApiError("FILE_TOO_LARGE", "CSV files must be 5 MB or smaller.", Map.of()));
    }

    @ExceptionHandler(Exception.class)
    ResponseEntity<ApiDtos.ApiError> handleUnexpected(Exception ex) {
        return ResponseEntity.status(HttpStatus.INTERNAL_SERVER_ERROR).body(
                new ApiDtos.ApiError("INTERNAL_ERROR", "The request could not be completed.", Map.of()));
    }
}

package com.datapulse.security;

import com.nimbusds.jose.jwk.source.ImmutableSecret;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Configuration;
import org.springframework.security.config.annotation.method.configuration.EnableMethodSecurity;
import org.springframework.security.config.annotation.web.builders.HttpSecurity;
import org.springframework.security.config.annotation.web.configuration.EnableWebSecurity;
import org.springframework.security.config.http.SessionCreationPolicy;
import org.springframework.security.crypto.bcrypt.BCryptPasswordEncoder;
import org.springframework.security.crypto.password.PasswordEncoder;
import org.springframework.security.oauth2.jose.jws.MacAlgorithm;
import org.springframework.security.oauth2.jwt.JwtDecoder;
import org.springframework.security.oauth2.jwt.JwtEncoder;
import org.springframework.security.oauth2.jwt.NimbusJwtDecoder;
import org.springframework.security.oauth2.jwt.NimbusJwtEncoder;
import org.springframework.security.oauth2.server.resource.authentication.JwtAuthenticationConverter;
import org.springframework.security.oauth2.server.resource.authentication.JwtGrantedAuthoritiesConverter;
import org.springframework.security.web.SecurityFilterChain;
import org.springframework.web.cors.CorsConfiguration;
import org.springframework.web.cors.CorsConfigurationSource;
import org.springframework.web.cors.UrlBasedCorsConfigurationSource;

import javax.crypto.SecretKey;
import javax.crypto.spec.SecretKeySpec;
import java.util.Base64;
import java.util.List;

@Configuration
@EnableWebSecurity
@EnableMethodSecurity
public class SecurityConfig {
    @Bean
    PasswordEncoder passwordEncoder() {
        return new BCryptPasswordEncoder(12);
    }

    @Bean
    SecretKey jwtSecretKey(@Value("${datapulse.jwt.secret-base64}") String secretBase64) {
        if (secretBase64 == null || secretBase64.isBlank()) {
            throw new IllegalStateException("DATAPULSE_JWT_SECRET_BASE64 must be set; generate one with openssl rand -base64 32");
        }
        byte[] keyBytes;
        try {
            keyBytes = Base64.getDecoder().decode(secretBase64.trim());
        } catch (IllegalArgumentException ex) {
            throw new IllegalStateException("DATAPULSE_JWT_SECRET_BASE64 must be valid Base64", ex);
        }
        if (keyBytes.length < 32) {
            throw new IllegalStateException("JWT HMAC secret must decode to at least 32 bytes");
        }
        return new SecretKeySpec(keyBytes, "HmacSHA256");
    }

    @Bean
    JwtEncoder jwtEncoder(SecretKey jwtSecretKey) {
        return new NimbusJwtEncoder(new ImmutableSecret<>(jwtSecretKey));
    }

    @Bean
    JwtDecoder jwtDecoder(SecretKey jwtSecretKey) {
        return NimbusJwtDecoder.withSecretKey(jwtSecretKey).macAlgorithm(MacAlgorithm.HS256).build();
    }

    @Bean
    JwtAuthenticationConverter jwtAuthenticationConverter() {
        JwtGrantedAuthoritiesConverter authorities = new JwtGrantedAuthoritiesConverter();
        authorities.setAuthoritiesClaimName("roles");
        authorities.setAuthorityPrefix("");
        JwtAuthenticationConverter converter = new JwtAuthenticationConverter();
        converter.setJwtGrantedAuthoritiesConverter(authorities);
        return converter;
    }

    @Bean
    SecurityFilterChain securityFilterChain(HttpSecurity http,
                                             JwtAuthenticationConverter jwtAuthenticationConverter,
                                             CorsConfigurationSource corsConfigurationSource) throws Exception {
        return http
                .csrf(csrf -> csrf.disable())
                .cors(cors -> cors.configurationSource(corsConfigurationSource))
                .sessionManagement(session -> session.sessionCreationPolicy(SessionCreationPolicy.STATELESS))
                .authorizeHttpRequests(auth -> auth
                        .requestMatchers("/api/auth/login", "/api/health").permitAll()
                        .anyRequest().authenticated())
                .oauth2ResourceServer(oauth -> oauth.jwt(jwt -> jwt.jwtAuthenticationConverter(jwtAuthenticationConverter)))
                .build();
    }

    @Bean
    CorsConfigurationSource corsConfigurationSource(@Value("${datapulse.cors.origin}") String origin) {
        CorsConfiguration configuration = new CorsConfiguration();
        configuration.setAllowedOrigins(List.of(origin));
        configuration.setAllowedMethods(List.of("GET", "POST", "PUT", "PATCH", "DELETE", "OPTIONS"));
        configuration.setAllowedHeaders(List.of("Authorization", "Content-Type"));
        configuration.setExposedHeaders(List.of("Location"));
        configuration.setAllowCredentials(false);
        UrlBasedCorsConfigurationSource source = new UrlBasedCorsConfigurationSource();
        source.registerCorsConfiguration("/api/**", configuration);
        return source;
    }
}

package com.datapulse.security;

import com.datapulse.domain.RoleEntity;
import com.datapulse.domain.UserEntity;
import com.datapulse.repository.RoleRepository;
import com.datapulse.repository.UserRepository;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.boot.ApplicationRunner;
import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Configuration;
import org.springframework.security.crypto.password.PasswordEncoder;

@Configuration
public class SeedUsers {
    @Bean
    ApplicationRunner seedDemoUsers(
            RoleRepository roles,
            UserRepository users,
            PasswordEncoder passwordEncoder,
            @Value("${datapulse.seed.admin-email:}") String adminEmail,
            @Value("${datapulse.seed.admin-password:}") String adminPassword,
            @Value("${datapulse.seed.analyst-email:}") String analystEmail,
            @Value("${datapulse.seed.analyst-password:}") String analystPassword) {
        return args -> {
            RoleEntity adminRole = roles.findByRoleName("ADMIN").orElseGet(() -> saveRole(roles, "ADMIN", "Manages all users and datasets."));
            RoleEntity analystRole = roles.findByRoleName("ANALYST").orElseGet(() -> saveRole(roles, "ANALYST", "Manages owned datasets and validation runs."));
            createIfConfigured(users, passwordEncoder, adminRole, adminEmail, adminPassword, "DataPulse Admin");
            createIfConfigured(users, passwordEncoder, analystRole, analystEmail, analystPassword, "DataPulse Analyst");
        };
    }

    private RoleEntity saveRole(RoleRepository roles, String name, String description) {
        RoleEntity role = new RoleEntity();
        role.setRoleName(name);
        role.setDescription(description);
        return roles.save(role);
    }

    private void createIfConfigured(UserRepository users, PasswordEncoder encoder, RoleEntity role,
                                    String email, String password, String fullName) {
        if (email == null || email.isBlank() || password == null || password.isBlank()) return;
        users.findByEmailIgnoreCase(email).orElseGet(() -> {
            UserEntity user = new UserEntity();
            user.setRole(role);
            user.setFullName(fullName);
            user.setEmail(email.trim().toLowerCase());
            user.setPasswordHash(encoder.encode(password));
            user.setActive(true);
            return users.save(user);
        });
    }

package com.datapulse.service;

import com.datapulse.api.ApiDtos;
import com.datapulse.api.ApiException;

import org.springframework.security.crypto.password.PasswordEncoder;
import org.springframework.security.oauth2.jose.jws.MacAlgorithm;
import org.springframework.security.oauth2.jwt.Jwt;
import org.springframework.security.oauth2.jwt.JwtClaimsSet;
import org.springframework.security.oauth2.jwt.JwtEncoder;
import org.springframework.security.oauth2.jwt.JwtEncoderParameters;
import org.springframework.security.oauth2.jwt.JwsHeader;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import java.time.Duration;
import java.time.Instant;
import java.util.List;

@Service
public class AuthService {
    private final UserRepository users;
    private final PasswordEncoder passwordEncoder;
    private final JwtEncoder jwtEncoder;

    public AuthService(UserRepository users, PasswordEncoder passwordEncoder, JwtEncoder jwtEncoder) {
        this.users = users;
        this.passwordEncoder = passwordEncoder;
        this.jwtEncoder = jwtEncoder;
    }

    @Transactional(readOnly = true)
    public ApiDtos.AuthResponse login(ApiDtos.LoginRequest request) {
        UserEntity user = users.findByEmailIgnoreCase(request.email().trim())
                .filter(UserEntity::isActive)
                .orElseThrow(() -> ApiException.unauthorized("Email or password is incorrect."));
        if (!passwordEncoder.matches(request.password(), user.getPasswordHash())) {
            throw ApiException.unauthorized("Email or password is incorrect.");
        }

        Instant now = Instant.now();
        JwtClaimsSet claims = JwtClaimsSet.builder()
                .issuer("datapulse")
                .issuedAt(now)
                .expiresAt(now.plus(Duration.ofHours(3)))
                .subject(user.getUserId().toString())
                .claim("email", user.getEmail())
                .claim("name", user.getFullName())
                .claim("roles", List.of("ROLE_" + user.getRole().getRoleName()))
                .build();
        JwsHeader header = JwsHeader.with(MacAlgorithm.HS256).build();
        String token = jwtEncoder.encode(JwtEncoderParameters.from(header, claims)).getTokenValue();
        return new ApiDtos.AuthResponse(token, userView(user));
    }

    @Transactional(readOnly = true)
    public ApiDtos.UserView currentUser(Jwt jwt) {
        long userId;
        try {
            userId = Long.parseLong(jwt.getSubject());
        } catch (NumberFormatException ex) {
            throw ApiException.unauthorized("The session is invalid. Please sign in again.");
        }
        UserEntity user = users.findById(userId)
                .filter(UserEntity::isActive)
                .orElseThrow(() -> ApiException.unauthorized("This account is unavailable."));
        return userView(user);
    }

    private ApiDtos.UserView userView(UserEntity user) {
        return new ApiDtos.UserView(user.getUserId(), user.getFullName(), user.getEmail(), user.getRole().getRoleName());
    }
}

@Service
public class CsvValidationService {
    private static final long MAX_BYTES = 5L * 1024 * 1024;
    private static final int MAX_ROWS = 5000;
    private static final Pattern EMAIL = Pattern.compile("^[A-Z0-9._%+-]+@[A-Z0-9.-]+\\.[A-Z]{2,}$", Pattern.CASE_INSENSITIVE);

    private final DatasetService datasetService;
    private final DatasetFieldRepository fields;
    private final ValidationRuleRepository rules;
    private final ValidationRunRepository runs;
    private final QualityIssueRepository issues;
    private final QualityScoreRepository scores;
    private final JdbcTemplate jdbcTemplate;
    private final EntityManager entityManager;
    private final RunQueryService runQueryService;

    public CsvValidationService(DatasetService datasetService, DatasetFieldRepository fields,
                                ValidationRuleRepository rules, ValidationRunRepository runs,
                                QualityIssueRepository issues, QualityScoreRepository scores,
                                JdbcTemplate jdbcTemplate, EntityManager entityManager,
                                RunQueryService runQueryService) {
        this.datasetService = datasetService;
        this.fields = fields;
        this.rules = rules;
        this.runs = runs;
        this.issues = issues;
        this.scores = scores;
        this.jdbcTemplate = jdbcTemplate;
        this.entityManager = entityManager;
        this.runQueryService = runQueryService;
    }

    @Transactional
    public ApiDtos.RunView run(long datasetId, long userId, boolean admin, MultipartFile file) {
        if (file == null || file.isEmpty()) throw ApiException.badRequest("Choose a CSV file to run validation.");
        if (file.getSize() > MAX_BYTES) throw ApiException.badRequest("CSV files must be 5 MB or smaller.");
        String filename = file.getOriginalFilename();
        if (filename == null || !filename.toLowerCase(Locale.ROOT).endsWith(".csv")) {
            throw ApiException.badRequest("Upload a file with the .csv extension.");
        }

        DatasetEntity dataset = datasetService.accessibleDataset(datasetId, userId, admin);
        if (!"ACTIVE".equals(dataset.getStatus())) {
            throw ApiException.badRequest("Archived datasets cannot start new validation runs.");
        }
        List<DatasetFieldEntity> datasetFields = fields.findAllByDataset_DatasetIdAndActiveTrueOrderByFieldIdAsc(datasetId);
        if (datasetFields.isEmpty()) throw ApiException.badRequest("Add at least one dataset field before running validation.");
        List<ValidationRuleEntity> activeRules = rules.findAllByField_Dataset_DatasetIdAndField_ActiveTrueAndActiveTrueOrderByRuleIdAsc(datasetId);
        if (activeRules.isEmpty()) throw ApiException.badRequest("Add at least one active validation rule before running validation.");

        ValidationRunEntity run = new ValidationRunEntity();
        run.setDataset(dataset);
        run.setStartedBy(dataset.getOwner().getUserId().equals(userId) ? dataset.getOwner() : findUser(userId));
        run.setStatus("RUNNING");
        run.setTotalRows(0);
        run.setIssueCount(0);
        run = runs.saveAndFlush(run);

        final ValidationRunEntity currentRun = run;
        setAuditActor(userId);
        List<QualityIssueEntity> foundIssues = new ArrayList<>();
        int rowCount;
        try (InputStreamReader reader = new InputStreamReader(file.getInputStream(), StandardCharsets.UTF_8);
             CSVParser parser = CSVFormat.DEFAULT.builder()
                     .setHeader()
                     .setSkipHeaderRecord(true)
                     .setIgnoreEmptyLines(true)
                     .setTrim(false)
                     .get()
                     .parse(reader)) {
            Map<String, String> headers = normalizedHeaders(parser.getHeaderMap().keySet());
            for (DatasetFieldEntity field : datasetFields) {
                String header = headers.get(field.getFieldName().toLowerCase(Locale.ROOT));
                if (header == null) throw ApiException.badRequest("CSV is missing the configured column: " + field.getFieldName());
            }
            Map<Long, Set<String>> uniqueValues = new HashMap<>();
            rowCount = 0;
            for (CSVRecord record : parser) {
                rowCount++;
                if (rowCount > MAX_ROWS) throw ApiException.badRequest("CSV files are limited to 5,000 data rows for this project demo.");
                for (ValidationRuleEntity rule : activeRules) {
                    DatasetFieldEntity field = rule.getField();
                    String header = headers.get(field.getFieldName().toLowerCase(Locale.ROOT));
                    String value = header != null && record.isMapped(header) && record.isSet(header) ? record.get(header).trim() : "";
                    String failure = failureMessage(rule, value, uniqueValues);
                    if (failure != null) {
                        QualityIssueEntity issue = new QualityIssueEntity();
                        issue.setRun(currentRun);
                        issue.setRule(rule);
                        issue.setRowNumber((int) record.getRecordNumber());
                        issue.setSeverity(rule.getSeverity());
                        issue.setMessage(failure);
                        issue.setObservedValue(value.isBlank() ? null : value);
                        foundIssues.add(issue);
                    }
                }
            }
        } catch (IOException | IllegalStateException ex) {
            throw ApiException.badRequest("Could not read the CSV. Check that it has one header row and valid CSV formatting.");
        }

        if (!foundIssues.isEmpty()) issues.saveAll(foundIssues);
        currentRun.setTotalRows(rowCount);
        currentRun.setIssueCount(foundIssues.size());
        currentRun.setStatus("COMPLETED");
        currentRun.setCompletedAt(OffsetDateTime.now());
        runs.save(currentRun);
        entityManager.flush();
        jdbcTemplate.update("CALL recalculate_quality_score(?)", currentRun.getRunId());
        entityManager.clear();
        return runQueryService.get(currentRun.getRunId(), userId, admin);
    }

    private com.datapulse.domain.UserEntity findUser(long userId) {
        return entityManager.getReference(com.datapulse.domain.UserEntity.class, userId);
    }

    private void setAuditActor(long userId) {
        jdbcTemplate.queryForObject("select set_config('app.user_id', ?, true)", String.class, Long.toString(userId));
    }

    private Map<String, String> normalizedHeaders(Set<String> rawHeaders) {
        if (rawHeaders == null || rawHeaders.isEmpty()) throw ApiException.badRequest("CSV must contain a header row.");
        Map<String, String> normalized = new HashMap<>();
        for (String header : rawHeaders) {
            if (header == null || header.isBlank()) throw ApiException.badRequest("CSV headers cannot be blank.");
            String key = header.trim().toLowerCase(Locale.ROOT);
            if (normalized.putIfAbsent(key, header) != null) throw ApiException.badRequest("CSV headers must be unique, ignoring case.");
        }
        return normalized;
    }

    private String failureMessage(ValidationRuleEntity rule, String value, Map<Long, Set<String>> uniqueValues) {
        return switch (rule.getRuleType()) {
            case "REQUIRED" -> value.isBlank() ? "A value is required." : null;
            case "UNIQUE" -> {
                if (value.isBlank()) yield null;
                Set<String> seen = uniqueValues.computeIfAbsent(rule.getRuleId(), ignored -> new HashSet<>());
                yield seen.add(value.toLowerCase(Locale.ROOT)) ? null : "This value duplicates an earlier row.";
            }
            case "EMAIL" -> value.isBlank() || EMAIL.matcher(value).matches() ? null : "Enter a valid email address.";
            case "RANGE" -> rangeFailure(rule, value);
            case "TYPE" -> typeFailure(rule.getField().getDataType(), value);
            default -> "This rule type is not supported by the validation engine.";
        };
    }

    private String typeFailure(String dataType, String value) {
        if (value.isBlank()) return null;
        boolean valid;
        try {
            valid = switch (dataType) {
                case "TEXT" -> true;
                case "INTEGER" -> value.matches("[+-]?\\d+");
                case "DECIMAL" -> { new BigDecimal(value); yield true; }
                case "DATE" -> { LocalDate.parse(value); yield true; }
                case "BOOLEAN" -> value.equalsIgnoreCase("true") || value.equalsIgnoreCase("false");
                default -> false;
            };
        } catch (RuntimeException ex) {
            valid = false;
        }
        return valid ? null : "Value is not a valid " + dataType.toLowerCase(Locale.ROOT) + ".";
    }

    private String rangeFailure(ValidationRuleEntity rule, String value) {
        if (value.isBlank()) return null;
        try {
            BigDecimal actual = new BigDecimal(value);
            BigDecimal min = new BigDecimal(rule.getRuleConfig().get("min").toString());
            BigDecimal max = new BigDecimal(rule.getRuleConfig().get("max").toString());
            return actual.compareTo(min) < 0 || actual.compareTo(max) > 0
                    ? "Value must be between " + min.stripTrailingZeros().toPlainString() + " and " + max.stripTrailingZeros().toPlainString() + "."
                    : null;
        } catch (RuntimeException ex) {
            return "Value must be a number within the configured range.";
        }
    }
}

    @Transactional(readOnly = true)
    public ApiDtos.DashboardView summary(long userId, boolean admin) {
        String accessClause = admin ? "" : " WHERE d.owner_id = ?";
        String aggregateSql = """
                SELECT count(*) AS dataset_count,
                       coalesce(sum(s.open_issue_count), 0) AS open_issue_count,
                       coalesce(round(avg(s.latest_overall_score), 2), 0) AS average_score
                  FROM dataset_quality_summary s
                  JOIN datasets d ON d.dataset_id = s.dataset_id
                """ + accessClause;
        Long[] aggregates = jdbcTemplate.query(aggregateSql, ps -> {
            if (!admin) ps.setLong(1, userId);
        }, rs -> {
            if (!rs.next()) return new Long[]{0L, 0L, 0L};
            BigDecimal average = rs.getBigDecimal("average_score");
            return new Long[]{rs.getLong("dataset_count"), rs.getLong("open_issue_count"),
                    average == null ? 0L : average.movePointRight(2).longValue()};
        });
        BigDecimal averageScore = BigDecimal.valueOf(aggregates[2], 2);

        String listSql = """
                SELECT s.dataset_id, s.dataset_name, s.dataset_status, s.latest_run_id,
                       s.latest_overall_score, s.open_issue_count
                  FROM dataset_quality_summary s
                  JOIN datasets d ON d.dataset_id = s.dataset_id
                """ + accessClause + " ORDER BY d.created_at DESC";
        List<ApiDtos.DashboardDataset> datasets = jdbcTemplate.query(listSql, ps -> {
            if (!admin) ps.setLong(1, userId);
        }, (rs, rowNum) -> new ApiDtos.DashboardDataset(
                rs.getLong("dataset_id"), rs.getString("dataset_name"), rs.getString("dataset_status"),
                (Long) rs.getObject("latest_run_id"), rs.getBigDecimal("latest_overall_score"),

    private void validateRule(String type, String severity, Map<String, Object> config) {
        if (!RULE_TYPES.contains(type)) throw ApiException.badRequest("Rule type must be REQUIRED, UNIQUE, EMAIL, RANGE, or TYPE.");
        if (!SEVERITIES.contains(severity)) throw ApiException.badRequest("Severity must be LOW, MEDIUM, or HIGH.");
        if ("RANGE".equals(type)) {
            if (config == null || !config.containsKey("min") || !config.containsKey("max")) {
                throw ApiException.badRequest("RANGE rules require numeric min and max settings.");
            }
            try {
                BigDecimal min = new BigDecimal(config.get("min").toString());
                BigDecimal max = new BigDecimal(config.get("max").toString());
                if (min.compareTo(max) > 0) throw ApiException.badRequest("RANGE min must not exceed max.");
            } catch (NumberFormatException ex) {
                throw ApiException.badRequest("RANGE min and max must be numbers.");
            }
        }
    }

    }

    private void validatePage(int page, int size) {
        if (page < 0 || size < 1 || size > 100) throw ApiException.badRequest("Page must be 0 or greater and size must be between 1 and 100.");
    }

    private String cleanQuery(String query) {
        return query == null || query.isBlank() ? null : query.trim();
    }

    private String normalizeStatus(String status) {
        if (status == null || status.isBlank()) return null;
        String value = status.trim().toUpperCase();
        if (!List.of("ACTIVE", "ARCHIVED").contains(value)) throw ApiException.badRequest("Status must be ACTIVE or ARCHIVED.");
        return value;
    }

    private String trimToNull(String value) {
        return value == null || value.isBlank() ? null : value.trim();
    }

    private void requireActive(DatasetEntity dataset) {
        if (!"ACTIVE".equals(dataset.getStatus())) throw ApiException.badRequest("Archived datasets cannot be modified.");
    }
}

package com.datapulse.service;

import com.datapulse.api.ApiDtos;
import com.datapulse.api.ApiException;
import com.datapulse.domain.IssueResolutionEntity;
import com.datapulse.domain.IssueStatus;
import com.datapulse.domain.QualityIssueEntity;
import com.datapulse.domain.UserEntity;
import com.datapulse.repository.IssueResolutionRepository;
import com.datapulse.repository.QualityIssueRepository;
import com.datapulse.repository.UserRepository;
import org.springframework.jdbc.core.JdbcTemplate;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import java.time.OffsetDateTime;
import java.util.Locale;

@Service
public class IssueService {
    private final QualityIssueRepository issues;
    private final IssueResolutionRepository resolutions;
    private final UserRepository users;
    private final DatasetService datasets;
    private final RunQueryService runQueries;
    private final JdbcTemplate jdbcTemplate;

    public IssueService(QualityIssueRepository issues, IssueResolutionRepository resolutions, UserRepository users,
                        DatasetService datasets, RunQueryService runQueries, JdbcTemplate jdbcTemplate) {
        this.issues = issues;
        this.resolutions = resolutions;
        this.users = users;
        this.datasets = datasets;
        this.runQueries = runQueries;
        this.jdbcTemplate = jdbcTemplate;
    }

    @Transactional
    public ApiDtos.IssueView updateStatus(long issueId, long userId, boolean admin, ApiDtos.StatusUpdateRequest request) {
        QualityIssueEntity issue = issues.findById(issueId).orElseThrow(() -> ApiException.notFound("Issue was not found."));
        datasets.accessibleDataset(issue.getRun().getDataset().getDatasetId(), userId, admin);
        IssueStatus nextStatus;
        try {
            nextStatus = IssueStatus.valueOf(request.status().trim().toUpperCase(Locale.ROOT));
        } catch (IllegalArgumentException ex) {
            throw ApiException.badRequest("Status must be OPEN, RESOLVED, or WAIVED.");
        }
        if (issue.getStatus() == nextStatus) throw ApiException.badRequest("Choose a different issue status.");

        jdbcTemplate.queryForObject("select set_config('app.user_id', ?, true)", String.class, Long.toString(userId));
        UserEntity actor = users.findById(userId).orElseThrow(() -> ApiException.unauthorized("User account was not found."));
        IssueResolutionEntity resolution = new IssueResolutionEntity();
        resolution.setIssue(issue);
        resolution.setResolvedBy(actor);
        resolution.setAction(switch (nextStatus) {
            case OPEN -> "REOPENED";
            case RESOLVED -> "FIXED";
            case WAIVED -> "WAIVED";
        });
        resolution.setNote(request.note().trim());
        resolutions.save(resolution);

        issue.setStatus(nextStatus);
        issue.setResolvedAt(nextStatus == IssueStatus.OPEN ? null : OffsetDateTime.now());
        issues.save(issue);
        return runQueries.issueView(issue);
    }
}

package com.datapulse.service;

import com.datapulse.api.ApiDtos;
import com.datapulse.api.ApiException;
import com.datapulse.domain.IssueResolutionEntity;
import com.datapulse.domain.QualityIssueEntity;
import com.datapulse.domain.QualityScoreEntity;
import com.datapulse.domain.ValidationRunEntity;
import com.datapulse.repository.IssueResolutionRepository;
import com.datapulse.repository.QualityIssueRepository;
import com.datapulse.repository.QualityScoreRepository;
import com.datapulse.repository.ValidationRunRepository;
import org.springframework.data.domain.Page;
import org.springframework.data.domain.PageRequest;
import org.springframework.data.domain.Sort;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import java.util.List;

@Service
public class RunQueryService {
    private final ValidationRunRepository runs;
    private final QualityIssueRepository issues;
    private final IssueResolutionRepository resolutions;
    private final QualityScoreRepository scores;
    private final DatasetService datasetService;

    public RunQueryService(ValidationRunRepository runs, QualityIssueRepository issues,
                           IssueResolutionRepository resolutions, QualityScoreRepository scores,
                           DatasetService datasetService) {
        this.runs = runs;
        this.issues = issues;
        this.resolutions = resolutions;
        this.scores = scores;
        this.datasetService = datasetService;
    }

    @Transactional(readOnly = true)
    public ApiDtos.RunView get(long runId, long userId, boolean admin) {
        ValidationRunEntity run = runs.findById(runId).orElseThrow(() -> ApiException.notFound("Validation run was not found."));
        datasetService.accessibleDataset(run.getDataset().getDatasetId(), userId, admin);
        Page<QualityIssueEntity> issuePage = issues.findAllByRun_RunIdOrderByRowNumberAsc(
                runId, PageRequest.of(0, 100, Sort.by("rowNumber").ascending()));
        QualityScoreEntity score = scores.findByRun_RunId(runId).orElse(null);
        return new ApiDtos.RunView(run.getRunId(), run.getDataset().getDatasetId(),
                run.getDataset().getDatasetName(), run.getStatus(), run.getTotalRows(), run.getIssueCount(),
                run.getStartedAt(), run.getCompletedAt(), score == null ? null : toScore(score),
                issuePage.getContent().stream().map(this::toIssue).toList());
    }

    @Transactional(readOnly = true)
    public ApiDtos.PageView<ApiDtos.IssueView> listIssues(long runId, long userId, boolean admin, int page, int size) {
        if (page < 0 || size < 1 || size > 100) throw ApiException.badRequest("Page must be 0 or greater and size must be between 1 and 100.");
        ValidationRunEntity run = runs.findById(runId).orElseThrow(() -> ApiException.notFound("Validation run was not found."));
        datasetService.accessibleDataset(run.getDataset().getDatasetId(), userId, admin);
        Page<QualityIssueEntity> result = issues.findAllByRun_RunIdOrderByRowNumberAsc(
                runId, PageRequest.of(page, size, Sort.by("rowNumber").ascending()));
        return new ApiDtos.PageView<>(result.getContent().stream().map(this::toIssue).toList(),
                result.getNumber(), result.getSize(), result.getTotalElements(), result.getTotalPages());
    }

    @Transactional(readOnly = true)
    public ApiDtos.IssueView issueView(QualityIssueEntity issue) {
        return toIssue(issue);
    }

    private ApiDtos.ScoreView toScore(QualityScoreEntity score) {
        return new ApiDtos.ScoreView(score.getCompletenessScore(), score.getValidityScore(),
                score.getUniquenessScore(), score.getOverallScore(), score.getCalculatedAt());
    }

    private ApiDtos.IssueView toIssue(QualityIssueEntity issue) {
        List<ApiDtos.ResolutionView> history = resolutions.findAllByIssue_IssueIdOrderByResolvedAtDesc(issue.getIssueId())
                .stream().map(this::toResolution).toList();
        return new ApiDtos.IssueView(issue.getIssueId(), issue.getRun().getRunId(), issue.getRule().getRuleId(),
                issue.getRule().getField().getFieldName(), issue.getRowNumber(), issue.getSeverity(), issue.getMessage(),
                issue.getObservedValue(), issue.getStatus().name(), issue.getCreatedAt(), issue.getResolvedAt(), history);
    }

    private ApiDtos.ResolutionView toResolution(IssueResolutionEntity resolution) {
        return new ApiDtos.ResolutionView(resolution.getResolutionId(), resolution.getAction(), resolution.getNote(),
                resolution.getResolvedBy().getFullName(), resolution.getResolvedAt());
    }
}
